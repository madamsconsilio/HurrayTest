        const FLOODGATE_URL = "https://floodgate.g.apple.com/api/anthropic/v1/messages";
        const DEFAULT_MODEL = "anthropic.claude-sonnet-4-6";

        // Load token from browser session storage on page load
        window.addEventListener("DOMContentLoaded", () => {
            const saved = sessionStorage.getItem("FloodgateIdToken");
            const status = document.getElementById("clientTokenStatus");
            const input = document.getElementById("txtClientToken");

            if (saved) {
                input.value = saved;
                status.innerHTML = "<span style='color:green;'>✓ Token active in browser session</span>";
            }
        });

        function saveTokenToBrowser() {
            const input = document.getElementById("txtClientToken");
            const status = document.getElementById("clientTokenStatus");
            const token = input.value.trim();

            if (!token) {
                sessionStorage.removeItem("FloodgateIdToken");
                status.innerHTML = "<span style='color:red;'>Token cannot be empty.</span>";
                return;
            }

            sessionStorage.setItem("FloodgateIdToken", token);
            status.innerHTML = "<span style='color:green;'>✓ Token saved in browser!</span>";
        }

        function toggleTokenVisibility() {
            const input = document.getElementById("txtClientToken");
            input.type = input.type === "password" ? "text" : "password";
        }

        async function testFloodgateDirectly() {
            const respBox = document.getElementById("floodgateDirectResponse");
            respBox.style.display = "block";
            respBox.style.color = "#333";
            respBox.innerHTML = "Sending request directly to Floodgate via your browser...";

            const token = document.getElementById("txtClientToken").value.trim() || sessionStorage.getItem("FloodgateIdToken");

            if (!token) {
                respBox.style.color = "red";
                respBox.innerHTML = "Error: Please enter a Floodgate token above first.";
                return;
            }

            const payload = {
                model: DEFAULT_MODEL,
                max_tokens: 1024,
                messages: [
                    { role: "user", content: "Hello! Please reply with a short confirmation message indicating that you received this request from the Hurray Relativity Custom Page browser bridge." }
                ]
            };

            try {
                const response = await fetch(FLOODGATE_URL, {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json",
                        "Authorization": `Bearer ${token}`
                    },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    const errText = await response.text();
                    respBox.style.color = "red";
                    respBox.innerHTML = `<strong>Floodgate returned HTTP ${response.status}:</strong><br/>${escapeHtml(errText)}`;
                    return;
                }

                const data = await response.json();
                let output = "";
                if (data.content && data.content.length > 0) {
                    output = data.content[0].text;
                } else {
                    output = JSON.stringify(data, null, 2);
                }

                respBox.style.color = "#03543f";
                respBox.innerHTML = `<strong>Floodgate Response (via Browser):</strong><br/>${escapeHtml(output)}`;
            } catch (err) {
                respBox.style.color = "red";
                respBox.innerHTML = `<strong>Browser Request Failed:</strong> ${escapeHtml(err.message)}<br/><br/>
                    <em>Ensure you are connected to Apple VPN and that floodgate.g.apple.com resolves on your Mac.</em>`;
            }
        }

        function escapeHtml(string) {
            return String(string)
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }

2. Default.aspx.cs (Clean Server Code Without Network Outbound Calls)

Replace Default.aspx.cs with this:

using Relativity.API;
using Relativity.CustomPages;
using Relativity.Services.Folder;
using Relativity.Services.Objects;
using Relativity.Services.Objects.DataContracts;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace Hurray2
{
    public partial class Default : Page
    {
        // Alternate way to get workspaceID
        private int WorkspaceArtifactId => ConnectionHelper.Helper().GetActiveCaseID();

        protected void Page_Load(object sender, EventArgs e)
        {
            try
            {
                var userInfo = ConnectionHelper.Helper().GetAuthenticationManager().UserInfo;
                int userArtifactId = userInfo.ArtifactID;
                string userEmailAddress = userInfo.EmailAddress;
                string firstName = userInfo.FirstName;
                string lastName = userInfo.LastName;
                string fullName = userInfo.FullName;
                int currentUserWorkspaceArtifactId = userInfo.WorkspaceUserArtifactID;

                string displayString = string.Format("User: {0}\r\nUser ID: {1}\r\nUser Email: {2}\r\n\r\n Hello {3}, the time is {4}.", fullName, userArtifactId, userEmailAddress, firstName, DateTime.Now.ToShortTimeString());
                string v = $"Active Workspace ID: {WorkspaceArtifactId}<br/>" + displayString.Replace("\r\n", "<br/>");

                if (!IsPostBack)
                {
                    if (DisplayLabel != null)
                    {
                        DisplayLabel.Text = v;
                    }

                    RegisterAsyncTask(new PageAsyncTask(PopulateFolderDropdownAsync));
                    RegisterAsyncTask(new PageAsyncTask(PopulatePrivFieldsDropdownAsync));
                    RegisterAsyncTask(new PageAsyncTask(PopulateHurrayProjectsDropdownAsync));
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:red;'>Page_Load Error: {ex.Message}</span>";
                }
            }
        }

        // -------------------------------------------------------------
        // BUTTON CLICK HANDLER (RELATIVITY OBJECT MANAGER)
        // -------------------------------------------------------------
        protected void btnGetControlNumbers_Click(object sender, EventArgs e)
        {
            RegisterAsyncTask(new PageAsyncTask(DisplayTop10ControlNumbersAsync));
        }

        private async Task DisplayTop10ControlNumbersAsync()
        {
            try
            {
                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
                {
                    var queryRequest = new QueryRequest
                    {
                        ObjectType = new ObjectTypeRef { ArtifactTypeID = 10 }, // 10 = Document
                        Fields = new List<FieldRef>
                        {
                            new FieldRef { Name = "Control Number" }
                        },
                        Sorts = new List<Sort>
                        {
                            new Sort
                            {
                                Direction = SortEnum.Ascending,
                                FieldIdentifier = new FieldRef { Name = "Control Number" }
                            }
                        }
                    };

                    QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 10);

                    if (result?.Objects != null && result.Objects.Count > 0)
                    {
                        List<string> controlNumbers = new List<string>();

                        foreach (RelativityObject doc in result.Objects)
                        {
                            string controlNumber = doc.FieldValues?
                                .FirstOrDefault(f => f.Field.Name == "Control Number")?.Value?.ToString()
                                ?? doc.Name;

                            controlNumbers.Add(controlNumber);
                        }

                        if (lblControlNumbers != null)
                        {
                            lblControlNumbers.Text = string.Join("<br/>", controlNumbers);
                        }
                    }
                    else if (lblControlNumbers != null)
                    {
                        lblControlNumbers.Text = "No documents found.";
                    }
                }
            }
            catch (Exception ex)
            {
                if (lblControlNumbers != null)
                {
                    lblControlNumbers.Text = $"<span style='color:red;'>Document Query Error: {ex.Message}</span>";
                }
            }
        }

        // -------------------------------------------------------------
        // FOLDER DROPDOWN
        // -------------------------------------------------------------
        private async Task PopulateFolderDropdownAsync()
        {
            try
            {
                if (ddlDynamicOptions == null) return;

                ddlDynamicOptions.Items.Clear();
                ddlDynamicOptions.Items.Add(new ListItem("-- Select a Folder --", ""));

                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IFolderManager folderManager = servicesMgr.CreateProxy<IFolderManager>(ExecutionIdentity.CurrentUser))
                {
                    List<Folder> rootfolders = await folderManager.GetFolderTreeAsync(WorkspaceArtifactId, new List<int>());
                    if (rootfolders != null)
                    {
                        foreach (Folder rootFolder in rootfolders)
                        {
                            AddFolderToDropDownList(rootFolder, level: 0);
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:orange;'>Folder Dropdown Error: {ex.Message}</span>";
                }
            }
        }

        private void AddFolderToDropDownList(Folder currentFolder, int level)
        {
            string indent = level == 0 ? string.Empty : new string('\u00A0', level * 4) + "└─ ";
            var item = new ListItem
            {
                Text = $"{indent}{currentFolder.Name}",
                Value = currentFolder.ArtifactID.ToString()
            };

            ddlDynamicOptions.Items.Add(item);

            if (currentFolder.Children != null && currentFolder.Children.Count > 0)
            {
                foreach (Folder child in currentFolder.Children)
                {
                    AddFolderToDropDownList(child, level + 1);
                }
            }
        }

        // -------------------------------------------------------------
        // FIELDS DROPDOWN
        // -------------------------------------------------------------
        private async Task PopulatePrivFieldsDropdownAsync()
        {
            try
            {
                if (ddlFields == null) return;

                ddlFields.Items.Clear();
                ddlFields.Items.Add(new ListItem("-- Select a PRIV Field --", ""));

                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
                {
                    var queryRequest = new QueryRequest
                    {
                        ObjectType = new ObjectTypeRef { ArtifactTypeID = 14 }, // 14 = Field
                        Condition = "'Object Type Artifact Type ID' == 10 AND 'Name' LIKE 'PRIV'",
                        Fields = new List<FieldRef>
                        {
                            new FieldRef { Name = "Name" }
                        },
                        Sorts = new List<Sort>
                        {
                            new Sort
                            {
                                Direction = SortEnum.Ascending,
                                FieldIdentifier = new FieldRef { Name = "Name" }
                            }
                        }
                    };

                    QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 1000);

                    if (result?.Objects != null)
                    {
                        foreach (RelativityObject fieldObj in result.Objects)
                        {
                            string fieldName = fieldObj.FieldValues?.Find(f => f.Field.Name == "Name")?.Value?.ToString();

                            if (string.IsNullOrEmpty(fieldName))
                            {
                                fieldName = !string.IsNullOrEmpty(fieldObj.Name)
                                    ? fieldObj.Name
                                    : $"Field {fieldObj.ArtifactID}";
                            }

                            string fieldArtifactId = fieldObj.ArtifactID.ToString();
                            ddlFields.Items.Add(new ListItem(fieldName, fieldArtifactId));
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:orange;'>PRIV Field Dropdown Error: {ex.Message}</span>";
                }
            }
        }

        // -------------------------------------------------------------
        // HURRAY PROJECTS DROPDOWN
        // -------------------------------------------------------------
        private async Task PopulateHurrayProjectsDropdownAsync()
        {
            try
            {
                if (ddlHurrayProjects == null) return;

                ddlHurrayProjects.Items.Clear();
                ddlHurrayProjects.Items.Add(new ListItem("-- Select a Hurray Project --", ""));

                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
                {
                    var queryRequest = new QueryRequest
                    {
                        ObjectType = new ObjectTypeRef { Name = "Hurray" },
                        Fields = new List<FieldRef>
                        {
                            new FieldRef { Name = "Project Name" }
                        },
                        Sorts = new List<Sort>
                        {
                            new Sort
                            {
                                Direction = SortEnum.Ascending,
                                FieldIdentifier = new FieldRef { Name = "Project Name" }
                            }
                        }
                    };

                    QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 1000);

                    if (result?.Objects != null && result.Objects.Count > 0)
                    {
                        foreach (RelativityObject projectObj in result.Objects)
                        {
                            string projectName = projectObj.FieldValues?
                                .FirstOrDefault(f => f.Field.Name == "Project Name")?.Value?.ToString();

                            if (string.IsNullOrWhiteSpace(projectName))
                            {
                                projectName = !string.IsNullOrWhiteSpace(projectObj.Name)
                                    ? projectObj.Name
                                    : $"Project {projectObj.ArtifactID}";
                            }

                            string artifactId = projectObj.ArtifactID.ToString();
                            ddlHurrayProjects.Items.Add(new ListItem(projectName, artifactId));
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:orange;'>Hurray Projects Dropdown Error: {ex.Message}</span>";
                }
            }
        }
    }
}
