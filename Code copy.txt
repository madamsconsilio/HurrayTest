        private const string ClaudeModel = "claude-3-5-sonnet-20241022";

        private int WorkspaceArtifactId => ConnectionHelper.Helper().GetActiveCaseID();

        {

            string firstName = Relativity.CustomPages.ConnectionHelper.Helper().GetAuthenticationManager().UserInfo.FirstName;
            string lastName = Relativity.CustomPages.ConnectionHelper.Helper().GetAuthenticationManager().UserInfo.LastName;
            string fullName = Relativity.CustomPages.ConnectionHelper.Helper().GetAuthenticationManager().UserInfo.FullName;
            int currentUserWorkspaceArtifactId = Relativity.CustomPages.ConnectionHelper.Helper().GetAuthenticationManager().UserInfo.WorkspaceUserArtifactID;

            string displayString = String.Format("User: {0}\r\nUser ID: {1}\r\nUser Email: {2}\r\n\r\n Hello {3}, the time is {4}.",
                fullName, userArtifactId, userEmailAddress, firstName, DateTime.Now.ToShortTimeString());
            string v = $"Active Workspace ID: {WorkspaceArtifactId}\r\n" + displayString;

            if (!IsPostBack)
            {
                DisplayLabel.Text = v;

                RegisterAsyncTask(new PageAsyncTask(PopulateFolderDropdownAsync));
                RegisterAsyncTask(new PageAsyncTask(PopulatePrivFieldsDropdownAsync));
                RegisterAsyncTask(new PageAsyncTask(PopulateHurrayProjectsDropdownAsync));
            }
        }

        // =========================================================================
        // BUTTON CLICK HANDLER: Analyze documents via Claude API
        // =========================================================================
        protected void btnGetControlNumbers_Click(object sender, EventArgs e)
        {
            RegisterAsyncTask(new PageAsyncTask(ProcessAndAnalyzeDocumentsAsync));
        }

        private async Task ProcessAndAnalyzeDocumentsAsync()
        {
            IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

            using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
            {
                // Optional: filter by selected folder if chosen
                string condition = "";
                if (!string.IsNullOrEmpty(ddlDynamicOptions.SelectedValue) && int.TryParse(ddlDynamicOptions.SelectedValue, out int folderId))
                {
                    condition = $"'Folder' == OBJECT {folderId}";
                }

                var queryRequest = new QueryRequest
                {
                    ObjectType = new ObjectTypeRef { ArtifactTypeID = 10 }, // 10 = Document
                    Condition = condition,
                    Fields = new List<FieldRef>
                    {
                        new FieldRef { Name = "Control Number" },
                        new FieldRef { Name = "Extracted Text" }
                    },
                    Sorts = new List<Sort>
                    {
                        new Sort
                        {
                            Direction = SortEnum.Ascending,
                            FieldIdentifier = new FieldRef { Name = "Control Number" }
                        }
                    }
                };

                // Query for top 10 documents
                QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 10);

                if (result?.Objects == null || result.Objects.Count == 0)
                {
                    lblControlNumbers.Text = "No documents found to analyze.";
                    return;
                }

                string projectName = ddlHurrayProjects.SelectedItem != null && !string.IsNullOrEmpty(ddlHurrayProjects.SelectedValue)
                    ? ddlHurrayProjects.SelectedItem.Text
                    : "General Review";

                List<string> displayResults = new List<string>();

                foreach (RelativityObject doc in result.Objects)
                {
                    string controlNumber = doc.FieldValues?.FirstOrDefault(f => f.Field.Name == "Control Number")?.Value?.ToString() ?? doc.Name;
                    string extractedText = doc.FieldValues?.FirstOrDefault(f => f.Field.Name == "Extracted Text")?.Value?.ToString() ?? string.Empty;

                    // Truncate to avoid exceeding token limits (Hurray standard: 30,000 characters)
                    if (extractedText.Length > 30000)
                    {
                        extractedText = extractedText.Substring(0, 30000);
                    }

                    try
                    {
                        // Call Claude API for document analysis
                        JObject analysisResult = await CallClaudeForPrivilegeAnalysisAsync(controlNumber, extractedText, projectName);

                        string status = analysisResult["status"]?.ToString() ?? "Unknown";
                        string rationale = analysisResult["rationale"]?.ToString() ?? "";
                        string narrative = analysisResult["privilege_log_narrative"]?.ToString() ?? "";

                        displayResults.Add($"<b>{controlNumber}</b>: {status}<br/>" +
                                           $"<i>Rationale:</i> {rationale}<br/>" +
                                           (!string.IsNullOrEmpty(narrative) ? $"<i>Priv Log:</i> {narrative}<br/>" : "") +
                                           "<hr/>");
                    }
                    catch (Exception ex)
                    {
                        displayResults.Add($"<b>{controlNumber}</b>: Error during analysis - {ex.Message}<hr/>");
                    }
                }

                lblControlNumbers.Text = string.Join("", displayResults);
            }
        }

        // =========================================================================
        // CLAUDE API INTEGRATION (Hurray Tool-Calling Pattern)
        // =========================================================================
        private async Task<JObject> CallClaudeForPrivilegeAnalysisAsync(string controlNumber, string extractedText, string projectName)
        {
            // Hurray System Prompt definition
            string systemPrompt = "You are an expert attorney performing second-pass privilege review on discovery documents. " +
                                  "Analyze whether attorney-client privilege (ACP), work product (WP), or other protections apply. " +
                                  "Determine privilege status, confidence score, rationale, flags, and generate privilege log entries. " +
                                  $"Matter/Project Name: {projectName}";

            string userPrompt = $"### Document Metadata\n- **Control Number:** {controlNumber}\n\n" +
                                $"### Document Content\n```\n{extractedText}\n```";

            // Payload structured matching Hurray's ClaudeRequest & record_privilege_analysis tool
            var requestBody = new
            {
                model = ClaudeModel,
                max_tokens = 2048,
                system = new[]
                {
                    new { type = "text", text = systemPrompt }
                },
                tools = new[]
                {
                    new
                    {
                        name = "record_privilege_analysis",
                        description = "Records the privilege determination, rationale, and privilege log entry for a document.",
                        input_schema = new
                        {
                            type = "object",
                            properties = new
                            {
                                status = new
                                {
                                    type = "string",
                                    @enum = new[] { "Privileged", "Not Privileged", "Potentially Privileged" },
                                    description = "Core privilege determination"
                                },
                                privilege_types = new
                                {
                                    type = "array",
                                    items = new { type = "string", @enum = new[] { "ACP", "WP", "Common Interest", "Mediation Privilege" } }
                                },
                                redact_privilege_tag = new
                                {
                                    type = "string",
                                    @enum = new[] { "Withhold", "Redaction Needed" }
                                },
                                confidence_score = new { type = "number", description = "0.0 to 1.0" },
                                rationale = new { type = "string", description = "Detailed legal reasoning" },
                                privilege_log_narrative = new { type = "string", description = "Privilege log prose description" }
                            },
                            required = new[] { "status", "rationale" }
                        }
                    }
                },
                tool_choice = new
                {
                    type = "tool",
                    name = "record_privilege_analysis"
                },
                messages = new[]
                {
                    new { role = "user", content = userPrompt }
                }
            };

            var requestMessage = new HttpRequestMessage(HttpMethod.Post, "https://api.anthropic.com/v1/messages");
            requestMessage.Headers.Add("x-api-key", ClaudeApiKey);
            requestMessage.Headers.Add("anthropic-version", "2023-06-01");
            requestMessage.Content = new StringContent(JsonConvert.SerializeObject(requestBody), Encoding.UTF8, "application/json");

            HttpResponseMessage response = await _httpClient.SendAsync(requestMessage);
            string responseJson = await response.Content.ReadAsStringAsync();

            if (!response.IsSuccessStatusCode)
            {
                throw new Exception($"Claude API error ({response.StatusCode}): {responseJson}");
            }

            JObject parsedResponse = JObject.Parse(responseJson);

            // Extract input parameters from the record_privilege_analysis tool_use block
            JToken toolUseBlock = parsedResponse["content"]?.FirstOrDefault(b => b["type"]?.ToString() == "tool_use");

            if (toolUseBlock != null && toolUseBlock["input"] is JObject inputObj)
            {
                return inputObj;
            }

            // Fallback if returned in raw text
            return new JObject
            {
                ["status"] = "Completed",
                ["rationale"] = parsedResponse["content"]?.FirstOrDefault(b => b["type"]?.ToString() == "text")?["text"]?.ToString() ?? "No output returned"
            };
        }

        // =========================================================================
        // DROPDOWN POPULATION METHODS
        // =========================================================================
        private async Task PopulateFolderDropdownAsync()
        {
            ddlDynamicOptions.Items.Clear();
            ddlDynamicOptions.Items.Add(new ListItem("-- Select a Folder --", ""));

            IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

            using (IFolderManager folderManager = servicesMgr.CreateProxy<IFolderManager>(ExecutionIdentity.CurrentUser))
            {
                List<Folder> rootfolders = await folderManager.GetFolderTreeAsync(WorkspaceArtifactId, new List<int>());
                if (rootfolders != null)
                {
                    foreach (Folder rootFolder in rootfolders)
                    {
                        AddFolderToDropDownList(rootFolder, level: 0);
                    }
                }
            }
        }

        private void AddFolderToDropDownList(Folder currentFolder, int level)
        {
            string indent = level == 0 ? string.Empty : new string('\u00A0', level * 4) + "└─ ";
            var item = new ListItem
            {
                Text = $"{indent}{currentFolder.Name}",
                Value = currentFolder.ArtifactID.ToString()
            };

            ddlDynamicOptions.Items.Add(item);

            if (currentFolder.Children != null && currentFolder.Children.Count > 0)
            {
                foreach (Folder child in currentFolder.Children)
                {
                    AddFolderToDropDownList(child, level + 1);
                }
            }
        }

        private async Task PopulatePrivFieldsDropdownAsync()
        {
            ddlFields.Items.Clear();
            ddlFields.Items.Add(new ListItem("-- Select a PRIV Field --", ""));

            IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

            using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
            {
                var queryRequest = new QueryRequest
                {
                    ObjectType = new ObjectTypeRef { ArtifactTypeID = 14 }, // 14 = Field
                    Condition = "'Object Type Artifact Type ID' == 10 AND 'Name' LIKE 'PRIV'",
                    Fields = new List<FieldRef>
                    {
                        new FieldRef { Name = "Name" }
                    },
                    Sorts = new List<Sort>
                    {
                        new Sort
                        {
                            Direction = SortEnum.Ascending,
                            FieldIdentifier = new FieldRef { Name = "Name" }
                        }
                    }
                };

                QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 1000);

                if (result?.Objects != null)
                {
                    foreach (RelativityObject fieldObj in result.Objects)
                    {
                        string fieldName = fieldObj.FieldValues?.Find(f => f.Field.Name == "Name")?.Value?.ToString();

                        if (string.IsNullOrEmpty(fieldName))
                        {
                            fieldName = !string.IsNullOrEmpty(fieldObj.Name)
                                ? fieldObj.Name
                                : $"Field {fieldObj.ArtifactID}";
                        }

                        string fieldArtifactId = fieldObj.ArtifactID.ToString();
                        ddlFields.Items.Add(new ListItem(fieldName, fieldArtifactId));
                    }
                }
            }
        }

        private async Task PopulateHurrayProjectsDropdownAsync()
        {
            ddlHurrayProjects.Items.Clear();
            ddlHurrayProjects.Items.Add(new ListItem("-- Select a Hurray Project --", ""));

            IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

            using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
            {
                var queryRequest = new QueryRequest
                {
                    ObjectType = new ObjectTypeRef { Name = "Hurray" },
                    Fields = new List<FieldRef>
                    {
                        new FieldRef { Name = "Project Name" }
                    },
                    Sorts = new List<Sort>
                    {
                        new Sort
                        {
                            Direction = SortEnum.Ascending,
                            FieldIdentifier = new FieldRef { Name = "Project Name" }
                        }
                    }
                };

                QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 1000);

                if (result?.Objects != null && result.Objects.Count > 0)
                {
                    foreach (RelativityObject projectObj in result.Objects)
                    {
                        string projectName = projectObj.FieldValues?
                            .FirstOrDefault(f => f.Field.Name == "Project Name")?.Value?.ToString();

                        if (string.IsNullOrWhiteSpace(projectName))
                        {
                            projectName = !string.IsNullOrWhiteSpace(projectObj.Name)
                                ? projectObj.Name
                                : $"Project {projectObj.ArtifactID}";
                        }

                        string artifactId = projectObj.ArtifactID.ToString();
                        ddlHurrayProjects.Items.Add(new ListItem(projectName, artifactId));
                    }
                }
            }
        }
