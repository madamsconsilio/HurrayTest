{1}\r\nUser Email: {2}\r\n\r\n Hello {3}, the time is {4}.", fullName, userArtifactId, userEmailAddress, firstName, DateTime.Now.ToShortTimeString());
                string v = $"Active Workspace ID: {WorkspaceArtifactId}<br/>" + displayString.Replace("\r\n", "<br/>");

                if (!IsPostBack)
                {
                    if (DisplayLabel != null) DisplayLabel.Text = v;

                    if (Session["FloodgateToken"] != null && lblTokenStatus != null)
                    {
                        lblTokenStatus.Text = "<span style='color:green;'>✓ Token active in session</span>";
                    }

            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:red;'>Page_Load Error: {ex.Message}</span>";
                }

        protected string CurrentFloodgateToken
        {
            get
            {
                if (txtFloodgateToken != null && !string.IsNullOrWhiteSpace(txtFloodgateToken.Text))
                {
                    return txtFloodgateToken.Text.Trim();
                }
                return Session["FloodgateToken"] as string;
            }
        }

        protected void btnSaveToken_Click(object sender, EventArgs e)
        {
            try
            {
                string token = txtFloodgateToken != null ? txtFloodgateToken.Text.Trim() : string.Empty;
                if (!string.IsNullOrEmpty(token))
                {
                    Session["FloodgateToken"] = token;
                    if (lblTokenStatus != null)
                    {
                        lblTokenStatus.Text = "<span style='color:green;'>✓ Token saved to session!</span>";
                    }
                }
                else
                {
                    Session["FloodgateToken"] = null;
                    if (lblTokenStatus != null)
                    {
                        lblTokenStatus.Text = "<span style='color:red;'>Token cannot be empty.</span>";
                    }
                }
            }
            catch (Exception ex)
            {
                if (lblTokenStatus != null)
                {
                    lblTokenStatus.Text = $"<span style='color:red;'>Error saving token: {ex.Message}</span>";
                }
            }
        }

        protected void btnGetControlNumbers_Click(object sender, EventArgs e)
        {
            RegisterAsyncTask(new PageAsyncTask(DisplayTop10ControlNumbersAsync));
        }

        private async Task DisplayTop10ControlNumbersAsync()
        {
            try
            {
                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
                {
                    var queryRequest = new QueryRequest
                    {
                        ObjectType = new ObjectTypeRef { ArtifactTypeID = 10 }, // 10 = Document
                        Fields = new List<FieldRef>
                        {
                            new FieldRef { Name = "Control Number" }
                        },
                        Sorts = new List<Sort>
                        {
                            new Sort
                            {
                                Direction = SortEnum.Ascending,
                                FieldIdentifier = new FieldRef { Name = "Control Number" }
                            }
                        }
                    };

                    QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 10);

                    if (result?.Objects != null && result.Objects.Count > 0)
                    {
                        List<string> controlNumbers = new List<string>();

                        foreach (RelativityObject doc in result.Objects)
                        {
                            string controlNumber = doc.FieldValues?
                                .FirstOrDefault(f => f.Field.Name == "Control Number")?.Value?.ToString()
                                ?? doc.Name;

                            controlNumbers.Add(controlNumber);
                        }

                        if (lblControlNumbers != null)
                        {
                            lblControlNumbers.Text = string.Join("<br/>", controlNumbers);
                        }
                    }
                    else if (lblControlNumbers != null)
                    {
                        lblControlNumbers.Text = "No documents found.";
                    }
                }
            }
            catch (Exception ex)
            {
                if (lblControlNumbers != null)
                {
                    lblControlNumbers.Text = $"<span style='color:red;'>Document Query Error: {ex.Message}</span>";
                }
            }
        }

        private async Task PopulateFolderDropdownAsync()
        {
            try
            {
                if (ddlDynamicOptions == null) return;

                ddlDynamicOptions.Items.Clear();
                ddlDynamicOptions.Items.Add(new ListItem("-- Select a Folder --", ""));

                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IFolderManager folderManager = servicesMgr.CreateProxy<IFolderManager>(ExecutionIdentity.CurrentUser))
                {
                    List<Folder> rootfolders = await folderManager.GetFolderTreeAsync(WorkspaceArtifactId, new List<int>());
                    if (rootfolders != null)
                    {
                        foreach (Folder rootFolder in rootfolders)
                        {
                            AddFolderToDropDownList(rootFolder, level: 0);
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:orange;'>Folder Dropdown Error: {ex.Message}</span>";
                }
            }
        }

        private void AddFolderToDropDownList(Folder currentFolder, int level)
        {
            string indent = level == 0 ? string.Empty : new string('\u00A0', level * 4) + "└─ ";
            var item = new ListItem
            {
                Text = $"{indent}{currentFolder.Name}",
                Value = currentFolder.ArtifactID.ToString()
            };

            ddlDynamicOptions.Items.Add(item);

            if (currentFolder.Children != null && currentFolder.Children.Count > 0)
            {
                foreach (Folder child in currentFolder.Children)
                {
                    AddFolderToDropDownList(child, level + 1);
                }
            }
        }

        private async Task PopulatePrivFieldsDropdownAsync()
        {
            try
            {
                if (ddlFields == null) return;

                ddlFields.Items.Clear();
                ddlFields.Items.Add(new ListItem("-- Select a PRIV Field --", ""));

                IServicesMgr servicesMgr = ConnectionHelper.Helper().GetServicesManager();

                using (IObjectManager objectManager = servicesMgr.CreateProxy<IObjectManager>(ExecutionIdentity.CurrentUser))
                {
                    var queryRequest = new QueryRequest
                    {
                        ObjectType = new ObjectTypeRef { ArtifactTypeID = 14 }, // 14 = Field
                        Condition = "'Object Type Artifact Type ID' == 10 AND 'Name' LIKE 'PRIV'",
                        Fields = new List<FieldRef>
                        {
                            new FieldRef { Name = "Name" }
                        },
                        Sorts = new List<Sort>
                        {
                            new Sort
                            {
                                Direction = SortEnum.Ascending,
                                FieldIdentifier = new FieldRef { Name = "Name" }
                            }
                        }
                    };

                    QueryResult result = await objectManager.QueryAsync(WorkspaceArtifactId, queryRequest, 1, 1000);

                    if (result?.Objects != null)
                    {
                        foreach (RelativityObject fieldObj in result.Objects)
                        {
                            string fieldName = fieldObj.FieldValues?.Find(f => f.Field.Name == "Name")?.Value?.ToString();

                            if (string.IsNullOrEmpty(fieldName))
                            {
                                fieldName = !string.IsNullOrEmpty(fieldObj.Name)
                                    ? fieldObj.Name
                                    : $"Field {fieldObj.ArtifactID}";
                            }

                            string fieldArtifactId = fieldObj.ArtifactID.ToString();
                            ddlFields.Items.Add(new ListItem(fieldName, fieldArtifactId));
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                if (DisplayLabel != null)
                {
                    DisplayLabel.Text += $"<br/><span style='color:orange;'>PRIV Field Dropdown Error: {ex.Message}</span>";
                }
            }
        }
